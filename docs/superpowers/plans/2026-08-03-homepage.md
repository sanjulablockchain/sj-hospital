# St. Joseph Hospital Homepage Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the St. Joseph Hospital marketing homepage at `/` in this Next.js 16 App Router project, matching the reference design exactly while adding real responsive behavior, scroll-reveal, count-up stats, and scroll parallax.

**Architecture:** Server Components for all static content (routing shell, layout chrome, section content), with small `'use client'` leaves only where the browser is genuinely needed: scroll-reveal (`RevealOnScroll`), count-up stats (`AnimatedCounter`), scroll parallax (`useScrollParallax` + two Hero leaves), and the mobile nav toggle (`MobileNav`). No new npm dependencies.

**Tech Stack:** Next.js 16.2.11 (App Router), React 19.2.4, TypeScript strict, Tailwind CSS v4 (CSS-first, no `tailwind.config.js`), `next/font/google` (Plus Jakarta Sans + Sora), `next/image` for all photos.

## Global Constraints

- Follow `CLAUDE.md`'s feature-based folder architecture: `app/` is routing-only, real content lives in `src/features/home/`, shared chrome in `src/components/layout/`, shared primitives in `src/components/ui/`, shared hooks in `src/hooks/`, config in `src/config/`.
- Import via the `@/*` alias (already configured in `tsconfig.json`). No `../../..` relative chains.
- Server Components by default. Add `'use client'` only to the specific leaf that needs state/effects/browser APIs, never higher in the tree.
- Tailwind v4 is CSS-first: all theme tokens (colors, fonts) go in `src/app/globals.css` via `@theme inline`. Do not create a `tailwind.config.js`.
- No dependency additions. Do not add `framer-motion`, `clsx`, or any animation/utility library. Everything here is plain React + CSS.
- The character `—` (em dash) must not appear anywhere in shipped content. Use a comma, period, or `,`/`:` instead.
- No mojibake: use clean, correctly-encoded UTF-8 text and, for the star ratings, inline SVG stars (not unicode glyphs), to avoid any encoding ambiguity.
- Images: `public/images/logo.png`, `public/images/hero.jpg`, `public/images/welcome.jpg`, `public/images/doctors.jpg` already exist (prepared during brainstorming: the logo was decoded from the reference file's embedded PNG, and the other three are real photos sourced from sjhospital.lk). Do not re-fetch or replace them. Reference dimensions: logo 1248×386, hero 1640×390 (already cropped to remove baked-in marketing text), welcome 2000×1500, doctors 1400×2105.
- No automated tests are configured or requested for this project (`CLAUDE.md`: "No test runner is configured yet"; user confirmed manual verification only). Each task's verification step is: `npx tsc --noEmit`, `npm run lint`, and a manual check in the browser (dev server should already be running, see Task 1). A full `npm run build` happens once, in the final task.
- All motion (float animations, spin, reveal, count-up, parallax) must be disabled under `prefers-reduced-motion: reduce`.
- `next lint`/`tsc` must stay clean (no `any`, no unused vars/imports) throughout. This is a strict TypeScript project.

Start the dev server once, before Task 1, and leave it running for the rest of implementation:

```tsx
"use client";

import { useEffect, useRef, useState } from "react";
import type { ReactNode } from "react";

type RevealOnScrollProps = {
  children: ReactNode;
  delayMs?: number;
  className?: string;
};

export function RevealOnScroll({
  children,
  delayMs = 0,
  className = "",
}: RevealOnScrollProps) {
  const ref = useRef<HTMLDivElement>(null);
  const [isVisible, setIsVisible] = useState(false);

  useEffect(() => {
    const node = ref.current;
    if (!node) return;

    const prefersReducedMotion =
      typeof window.matchMedia === "function" &&
      window.matchMedia("(prefers-reduced-motion: reduce)").matches;

    if (prefersReducedMotion) {
      setIsVisible(true);
      return;
    }

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsVisible(true);
          observer.unobserve(node);
        }
      },
      { threshold: 0.15, rootMargin: "0px 0px -8% 0px" }
    );

    observer.observe(node);
    return () => observer.disconnect();
  }, []);

  return (
    <div
      ref={ref}
      className={`transition-all duration-700 ease-out ${
        isVisible ? "opacity-100 translate-y-0" : "opacity-0 translate-y-5"
      } ${className}`}
      style={{ transitionDelay: isVisible ? `${delayMs}ms` : "0ms" }}
    >
      {children}
    </div>
  );
}
```

```tsx
"use client";

import { useEffect, useRef, useState } from "react";

type AnimatedCounterProps = {
  target: number;
  prefix?: string;
  suffix?: string;
  durationMs?: number;
  className?: string;
};

export function AnimatedCounter({
  target,
  prefix = "",
  suffix = "",
  durationMs = 1400,
  className = "",
}: AnimatedCounterProps) {
  const ref = useRef<HTMLSpanElement>(null);
  const [value, setValue] = useState(0);
  const hasAnimated = useRef(false);

  useEffect(() => {
    const node = ref.current;
    if (!node) return;

    const prefersReducedMotion =
      typeof window.matchMedia === "function" &&
      window.matchMedia("(prefers-reduced-motion: reduce)").matches;

    if (prefersReducedMotion) {
      setValue(target);
      return;
    }

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting && !hasAnimated.current) {
          hasAnimated.current = true;
          const start = performance.now();

          const step = (now: number) => {
            const progress = Math.min((now - start) / durationMs, 1);
            const eased = 1 - Math.pow(1 - progress, 3);
            setValue(Math.round(target * eased));
            if (progress < 1) requestAnimationFrame(step);
          };

          requestAnimationFrame(step);
          observer.unobserve(node);
        }
      },
      { threshold: 0.4 }
    );

    observer.observe(node);
    return () => observer.disconnect();
  }, [target, durationMs]);

  return (
    <span ref={ref} className={className}>
      {prefix}
      {value.toLocaleString("en-US")}
      {suffix}
    </span>
  );
}
```

- [ ] **Step 3: Create `src/hooks/useScrollParallax.ts`**

```ts
"use client";

import { useEffect, useRef, useState } from "react";

/**
 * Tracks how far the attached element's center is from the viewport's
 * vertical center and returns a clamped translateY offset scaled by
 * `speed`, for a subtle scroll-linked parallax effect. Disabled under
 * prefers-reduced-motion.
 */
export function useScrollParallax(speed = 0.15, maxOffset = 60) {
  const ref = useRef<HTMLDivElement>(null);
  const [offset, setOffset] = useState(0);

  useEffect(() => {
    const node = ref.current;
    if (!node) return;

    const prefersReducedMotion =
      typeof window.matchMedia === "function" &&
      window.matchMedia("(prefers-reduced-motion: reduce)").matches;

    if (prefersReducedMotion) return;

    let ticking = false;

    const update = () => {
      const rect = node.getBoundingClientRect();
      const viewportCenter = window.innerHeight / 2;
      const distanceFromCenter = rect.top + rect.height / 2 - viewportCenter;
      const raw = distanceFromCenter * speed;
      setOffset(Math.max(-maxOffset, Math.min(maxOffset, raw)));
      ticking = false;
    };

    const onScrollOrResize = () => {
      if (!ticking) {
        ticking = true;
        requestAnimationFrame(update);
      }
    };

    update();
    window.addEventListener("scroll", onScrollOrResize, { passive: true });
    window.addEventListener("resize", onScrollOrResize);
    return () => {
      window.removeEventListener("scroll", onScrollOrResize);
      window.removeEventListener("resize", onScrollOrResize);
    };
  }, [speed, maxOffset]);

  return { ref, offset };
}
```
